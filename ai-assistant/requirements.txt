# AI Assistant — runtime requirements
#
# The pins below are ALIGNED with shared-canonical-services==2026.11.4
# (shared/pyproject.toml). The two are installed into the same image, so a
# conflicting pin (httpx 0.27 vs 0.28, ...) makes the install unsatisfiable.
#
# `shared-canonical-services` itself is NOT listed here: it is a PRIVATE
# wheel that is not on PyPI. deploy-ai.sh stages it into wheels/ and
# Dockerfile.prod installs requirements + wheel in ONE pip resolution.

# Web framework (the app imports fastapi directly; the wheel only carries it
# under its `dsar` extra, so pin it here, aligned with that extra)
fastapi==0.115.0
uvicorn[standard]==0.32.1
gunicorn==23.0.0
httpx==0.28.1
python-multipart==0.0.6

# Database drivers: the app runs on PostgreSQL and shared.database builds a
# postgresql:// (psycopg v3) engine at import + database_config imports psycopg2
# at module top. In wheel 2026.11.4 both are only in the 'database' extra, which
# the one-shot `pip wheel` in Dockerfile.prod does not install - without these
# pins the container crash-loops with ModuleNotFoundError: psycopg (2026-11-19).
# Versions match the wheel's database extra.
psycopg2-binary==2.9.13
psycopg[binary]==3.3.6
